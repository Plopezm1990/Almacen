function crearLogicaGastos({ setGastosGenerales }) {
  function addGasto(data) {
    setGastosGenerales((s2) => [...s2, { id: uid(), ...data }]);
  }
  function deleteGasto(id) {
    setGastosGenerales((s2) => s2.filter((g2) => g2.id !== id));
  }
  return { addGasto, deleteGasto };
}
