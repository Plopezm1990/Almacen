    if (!ventaId) return { ok: false, error: "Esa venta no tiene identificador y no se puede anular." };
    const lineas = (movimientosActuales || []).filter((m22) => m22.ventaId === ventaId || m22.operationId === ventaId).filter((m22) => esVenta(m22) || esSalida(m22)).filter((m22) => movimientoEsDelLocalActivoVenta(m22));
    if (lineas.length === 0) {
      return { ok: false, error: "No se han encontrado las l\xEDneas de esa venta." };
    }
    const yaAnulada = (movimientosActuales || []).some((m22) => m22.anulaVentaId === ventaId || m22.revierteMovimientoId && lineas.some((l22) => l22.id === m22.revierteMovimientoId));
    if (yaAnulada) {
      return { ok: false, error: "Esa venta ya estaba anulada." };
    }
    let aplicadas = 0;
    lineas.forEach((l22) => {
      const cantidadOriginal = Math.abs(cantidadConSigno(l22));
      if (cantidadOriginal <= 0) return;
      const r2 = aplicarMovimientoStock({
        productoId: l22.productoId,
        cantidad: cantidadOriginal,
        tipo: l22.tipo === "salida" ? "VENTA" : l22.tipo,
        operationId: uid(),
        movimientoId: uid(),
        origen: "anularVenta",
        documentoOrigenId: ventaId,
        afectaStockTotal: true,
        afectaStockPisoVenta: true,
        permitirDeficit: true,
        revierteMovimientoId: l22.id,
        motivo: `Anulaci\xF3n de venta${motivo ? " \xB7 " + motivo : ""}`,
        camposExtra: {
          anulaVentaId: ventaId,
          costoUnitario: l22.costoUnitario,
          // El ingreso se anota en negativo para que los informes de ventas
          // resten esta operación en vez de contarla como un ingreso nuevo.
          ingresoUnitario: l22.ingresoUnitario != null ? -Math.abs(Number(l22.ingresoUnitario)) : null,
          ivaVentaAplicado: l22.ivaVentaAplicado,
          medioPago: l22.medioPago,
          detallePago: l22.detallePago
        }
      });
      if (r2.ok) aplicadas++;
    });
    const fechaVenta = lineas[0]?.fecha;
    const arqueoDelDia = fechaVenta && (arqueos || []).find((a22) => a22.fecha === fechaVenta && (!localActivoId || a22.localId === localActivoId));
    if (arqueoDelDia && typeof window !== "undefined" && window.__nubeActiva) {
      fetch("https://flqercbgpgmmfaakrwkc.supabase.co/functions/v1/enviar-notificacion", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          titulo: "Caja cerrada, ahora desactualizada",
          cuerpo: `Se anul\xF3 una venta del ${fechaVenta}, d\xEDa que ya ten\xEDa la caja cerrada \u2014 revisa ese arqueo, puede que ya no cuadre.`,
          localId: lineas[0]?.localId || null,
          url: "/"
        })
      }).catch(() => {
      });
    }
    return { ok: true, lineasAnuladas: aplicadas, arqueoAfectado: !!arqueoDelDia };
  }
